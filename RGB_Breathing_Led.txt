/*
  YD-ESP32-S3 板载 RGB 呼吸灯程序
  适用于 ESP32-S3 开发板
  
  功能：
  - 红、绿、蓝三种颜色依次呼吸变化
  - 每种颜色从暗到亮，再从亮到暗
  - 颜色平滑过渡，循环播放
  
  板载 RGB LED：WS2812 类型，连接 GPIO 48
*/

#include <Adafruit_NeoPixel.h>

#define LED_PIN    48
#define LED_COUNT  1

Adafruit_NeoPixel pixels(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

#define FADE_STEP  5
#define FADE_DELAY 20

void setup() {
  pixels.begin();
  pixels.setBrightness(255);
  pixels.clear();
  pixels.show();
  
  Serial.begin(115200);
  delay(1000);
  Serial.println("YD-ESP32-S3 RGB Breathing LED Started!");
  Serial.println("WS2812 LED on GPIO 48");
}

void fadeColor(int r, int g, int b) {
  for (int brightness = 0; brightness <= 255; brightness += FADE_STEP) {
    int cr = map(brightness, 0, 255, 0, r);
    int cg = map(brightness, 0, 255, 0, g);
    int cb = map(brightness, 0, 255, 0, b);
    pixels.setPixelColor(0, pixels.Color(cr, cg, cb));
    pixels.show();
    delay(FADE_DELAY);
  }
  
  delay(200);
  
  for (int brightness = 255; brightness >= 0; brightness -= FADE_STEP) {
    int cr = map(brightness, 0, 255, 0, r);
    int cg = map(brightness, 0, 255, 0, g);
    int cb = map(brightness, 0, 255, 0, b);
    pixels.setPixelColor(0, pixels.Color(cr, cg, cb));
    pixels.show();
    delay(FADE_DELAY);
  }
}

void breathingRed() {
  Serial.println("Red breathing...");
  fadeColor(255, 0, 0);
}

void breathingGreen() {
  Serial.println("Green breathing...");
  fadeColor(0, 255, 0);
}

void breathingBlue() {
  Serial.println("Blue breathing...");
  fadeColor(0, 0, 255);
}

void breathingYellow() {
  Serial.println("Yellow breathing...");
  fadeColor(255, 255, 0);
}

void breathingPurple() {
  Serial.println("Purple breathing...");
  fadeColor(255, 0, 255);
}

void breathingCyan() {
  Serial.println("Cyan breathing...");
  fadeColor(0, 255, 255);
}

void breathingWhite() {
  Serial.println("White breathing...");
  fadeColor(255, 255, 255);
}

void loop() {
  breathingRed();
  delay(200);
  
  breathingGreen();
  delay(200);
  
  breathingBlue();
  delay(200);
  
  breathingYellow();
  delay(200);
  
  breathingPurple();
  delay(200);
  
  breathingCyan();
  delay(200);
  
  breathingWhite();
  delay(300);
}
