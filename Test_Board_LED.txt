/*
  板载LED测试程序
  适用于 ESP32-S3 开发板
  
  功能：
  - 测试板载LED是否正常工作
  - 每秒闪烁一次
  
  常用板载LED引脚：GPIO 2 或 GPIO 13
*/

#define LED_PIN 2

void setup() {
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  
  Serial.begin(115200);
  delay(1000);
  Serial.println("Board LED Test Started!");
  Serial.print("LED pin: ");
  Serial.println(LED_PIN);
}

void loop() {
  digitalWrite(LED_PIN, HIGH);
  Serial.println("LED ON");
  delay(500);
  
  digitalWrite(LED_PIN, LOW);
  Serial.println("LED OFF");
  delay(500);
}
