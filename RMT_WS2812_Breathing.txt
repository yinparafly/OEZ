/*
  YD-ESP32-S3 RMT 驱动 WS2812 RGB 呼吸灯
  纯硬件驱动，无需外部库
  
  功能：红→绿→蓝→黄→紫→青→白 循环呼吸
  
  与 NeoPixel 版本的区别：
  - 呼吸速度更快（FADE_STEP=10, FADE_DELAY=10）
  - 每种颜色呼吸两次
  - 方便区分两个版本
  
  WS2812 RGB LED 连接 GPIO 48
*/

#include <driver/rmt.h>

#define WS2812_PIN GPIO_NUM_48
#define RMT_CHANNEL RMT_CHANNEL_0
#define LED_COUNT 1

#define T0H 350
#define T0L 900
#define T1H 900
#define T1L 350

#define RMT_CLOCK_DIV 2
#define RMT_TICK_NS (1000000000LL / (APB_CLK_FREQ / RMT_CLOCK_DIV))

#define FADE_STEP  10
#define FADE_DELAY 10

rmt_item32_t led_data[LED_COUNT * 24];

void rmt_set_color(uint8_t r, uint8_t g, uint8_t b) {
  int item_idx = 0;
  
  for (int color = 0; color < 3; color++) {
    uint8_t byte_data;
    if (color == 0) byte_data = g;
    else if (color == 1) byte_data = r;
    else byte_data = b;
    
    for (int bit = 7; bit >= 0; bit--) {
      if (byte_data & (1 << bit)) {
        led_data[item_idx].level0 = 1;
        led_data[item_idx].duration0 = T1H / RMT_TICK_NS;
        led_data[item_idx].level1 = 0;
        led_data[item_idx].duration1 = T1L / RMT_TICK_NS;
      } else {
        led_data[item_idx].level0 = 1;
        led_data[item_idx].duration0 = T0H / RMT_TICK_NS;
        led_data[item_idx].level1 = 0;
        led_data[item_idx].duration1 = T0L / RMT_TICK_NS;
      }
      item_idx++;
    }
  }
  
  rmt_write_items(RMT_CHANNEL, led_data, LED_COUNT * 24, true);
}

void breathing_color(uint8_t r, uint8_t g, uint8_t b) {
  for (int breath = 0; breath < 2; breath++) {
    for (int brightness = 0; brightness <= 255; brightness += FADE_STEP) {
      int cr = map(brightness, 0, 255, 0, r);
      int cg = map(brightness, 0, 255, 0, g);
      int cb = map(brightness, 0, 255, 0, b);
      rmt_set_color(cr, cg, cb);
      delay(FADE_DELAY);
    }
    
    delay(100);
    
    for (int brightness = 255; brightness >= 0; brightness -= FADE_STEP) {
      int cr = map(brightness, 0, 255, 0, r);
      int cg = map(brightness, 0, 255, 0, g);
      int cb = map(brightness, 0, 255, 0, b);
      rmt_set_color(cr, cg, cb);
      delay(FADE_DELAY);
    }
    
    delay(100);
  }
}

void setup() {
  rmt_config_t config = RMT_DEFAULT_CONFIG_TX(WS2812_PIN, RMT_CHANNEL);
  config.clk_div = RMT_CLOCK_DIV;
  config.tx_config.loop_en = false;
  config.tx_config.carrier_en = false;
  config.tx_config.idle_level = RMT_IDLE_LEVEL_LOW;
  config.tx_config.idle_output_en = true;
  
  rmt_config(&config);
  rmt_driver_install(RMT_CHANNEL, 0, 0);
  
  Serial.begin(115200);
  delay(1000);
  Serial.println("==================================");
  Serial.println("YD-ESP32-S3 RMT WS2812 Breathing");
  Serial.println("Mode: Fast Double Breath");
  Serial.println("FADE_STEP=10, FADE_DELAY=10");
  Serial.println("Each color breathes twice");
  Serial.println("==================================");
}

void loop() {
  Serial.println("Red (x2)");
  breathing_color(255, 0, 0);
  
  Serial.println("Green (x2)");
  breathing_color(0, 255, 0);
  
  Serial.println("Blue (x2)");
  breathing_color(0, 0, 255);
  
  Serial.println("Yellow (x2)");
  breathing_color(255, 255, 0);
  
  Serial.println("Purple (x2)");
  breathing_color(255, 0, 255);
  
  Serial.println("Cyan (x2)");
  breathing_color(0, 255, 255);
  
  Serial.println("White (x2)");
  breathing_color(255, 255, 255);
  
  delay(200);
}
